cmake_minimum_required(VERSION 3.18)
project(auto_mode_manager LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/AutoModeStatus.msg" DEPENDENCIES builtin_interfaces)

option(BUILD_RUNTIME_NODES "Build manager runtime node" ON)
if(BUILD_RUNTIME_NODES)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_action REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(tf2 REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(tf2_geometry_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(robot_port REQUIRED)
  find_package(gn10_navigation REQUIRED)
  find_package(lidar_perception_system REQUIRED)
  find_package(field_target_pub REQUIRED)
  find_package(shooter_calculator REQUIRED)
  add_executable(auto_mode_manager_node src/auto_mode_manager_node.cpp)
  target_compile_options(auto_mode_manager_node PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(auto_mode_manager_node PRIVATE include)
  ament_target_dependencies(auto_mode_manager_node
    rclcpp rclcpp_action rcl_interfaces tf2 tf2_ros tf2_geometry_msgs geometry_msgs nav_msgs
    robot_port gn10_navigation lidar_perception_system field_target_pub shooter_calculator)
  rosidl_get_typesupport_target(cpp_typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
  target_link_libraries(auto_mode_manager_node "${cpp_typesupport}")
  install(TARGETS auto_mode_manager_node DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

# Local checks are deliberately not published with the package.
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/simple_control_test.cpp")
  enable_testing()
  add_executable(simple_control_test test/simple_control_test.cpp)
  target_include_directories(simple_control_test PRIVATE include)
  add_test(NAME simple_control_test COMMAND simple_control_test)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
